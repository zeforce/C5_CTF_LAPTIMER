cmake_minimum_required(VERSION 3.16)
if(NOT EXISTS "$ENV{IDF_PATH}/components/esp_phy/lib/esp32c5/libphy.a")
    message(FATAL_ERROR "ESP-IDF ESP32-C5 PHY archive not found; use ESP-IDF 6.0.2")
endif()
file(SHA256 "$ENV{IDF_PATH}/components/esp_phy/lib/esp32c5/libphy.a" DETECTOR_PHY_SHA256)
if(NOT DETECTOR_PHY_SHA256 STREQUAL "dbf33c418c8d408d4005c849d12a1432deea82e2e5e57de3c8ddf914d104fffb")
    message(FATAL_ERROR "Unverified ESP32-C5 libphy.a; use upstream C5VRX ESP-IDF 6.0.2 PHY")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(c5_detector)
if(C5_MINIMAL)
    add_compile_definitions(C5_MINIMAL=1)
endif()
